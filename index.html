THE WORK IS GROWING

A quick script is easy.
A fast research loop is harder.

One language makes the first analysis comfortable. Another may be needed when the model becomes a bottleneck. Moving between them means translating logic, maintaining two code paths, and slowing down the experiment.

You need a language that stays expressive while the calculations get serious. That's the problem Julia set out to tackle.

THE QUESTIONCan one language feel interactive and still compile efficient numerical code?

Julia.

A high-level, dynamically typed language designed for technical computing, with a just-in-time compiler that specializes code for the types it sees.

                    03 / LANGUAGE COMPARISON
                    Julia and C: the two-language problem
                  
                  +
                
                
    
                  
    
                    
    The same array update can be written at different levels: Julia states the elementwise operation, while C spells out the loop and destination storage.
    
                    
    
                      
    
                        JULIA / ARRAY BROADCAST
                        
    daily_change = rain .- evaporation
    
                      
    
                      
    
                        C / EXPLICIT ARRAY LOOP
                        
    #include <stddef.h>

void subtract_evaporation(const double *rain,
                          double evaporation,
                          double *daily_change,
                          size_t count)
{
    for (size_t day = 0; day < count; ++day) {
        daily_change[day] = rain[day] - evaporation;
    }
}
    
                      
    
                    
    
                    
    Julia's broadcast creates a result array. In C, the caller supplies writable output storage and the element count.
    
                    
    
                      THE TWO-LANGUAGE PROBLEM
                      
    Some scientific workflows prototype in a high-level language, then rewrite performance-critical parts in C or Fortran. Julia is designed to support high-level exploration and performance-oriented numerical code in one language. That's a design goal, not a speed guarantee: performance depends on the code and workload.

    """Dependency-free Python counterpart to garden_week.jl's soil-water model."""

import gc
import os
import platform
import statistics
import sys
import time


class Plant:
    __slots__ = ()


class Lettuce(Plant):
    __slots__ = ()


class Cactus(Plant):
    __slots__ = ()


def dry_limit(plant: Plant) -> float:
    # Julia selects a dry_limit method by plant type; this Python counterpart
    # expresses the same choice with explicit isinstance branches.
    if isinstance(plant, Lettuce):
        return 5.0
    if isinstance(plant, Cactus):
        return 2.0
    raise TypeError("Unknown plant")


def simulate(rain: list[float], plant: Plant, temperature: float):
    evaporation = 0.1 * temperature

    # Julia writes `rain .- evaporation`; built-in Python lists need this
    # comprehension to apply the same scalar subtraction to every rainfall value.
    daily_change = [value - evaporation for value in rain]

    # Julia's first array index is 1; Python's is 0, so day zero seeds the recurrence.
    soil = [0.0] * len(rain)
    initial_water = 10.0 + daily_change[0]
    soil[0] = min(20.0, max(0.0, initial_water))
    for day in range(1, len(rain)):
        water = soil[day - 1] + daily_change[day]
        if water < 0.0:
            water = 0.0
        elif water > 20.0:
            water = 20.0
        soil[day] = water

    limit = dry_limit(plant)
    # This counts the same threshold matches as Julia's count(predicate, soil).
    dry_days = sum(water < limit for water in soil)
    return soil, dry_days, dry_days <= 2


def timed_simulation(rain: list[float], plant: Plant, temperature: float):
    started = time.perf_counter_ns()
    result = simulate(rain, plant, temperature)
    elapsed = (time.perf_counter_ns() - started) / 1_000_000_000
    return elapsed, result


def main():
    days = int(os.environ.get("GARDEN_DAYS", "1000000"))
    repeats = int(os.environ.get("GARDEN_REPEATS", "5"))
    if days < 1:
        raise ValueError("GARDEN_DAYS must be positive")
    if repeats < 1 or repeats % 2 == 0:
        raise ValueError("GARDEN_REPEATS must be a positive odd number")

    # Match the deterministic inputs generated in garden_week.jl.
    rain = [((day * 37) % 500) / 100.0 for day in range(1, days + 1)]
    plant = Lettuce()
    temperature = 25.0
    jit = getattr(sys, "_jit", None)
    if jit is None or not hasattr(jit, "is_available") or not jit.is_available():
        native_jit_status = "unavailable"
    else:
        native_jit_status = "enabled" if jit.is_enabled() else "available, disabled"

    # In the measured CPython build the optional native JIT is unavailable.
    time.perf_counter_ns()
    gc.collect()
    started = time.perf_counter_ns()
    cold_result = simulate(rain, plant, temperature)
    cold_seconds = (time.perf_counter_ns() - started) / 1_000_000_000

    # Warm the interpreter path, then measure repeated runs after collecting garbage.
    timed_simulation(rain, plant, temperature)
    for _ in range(2):
        gc.collect()
        timed_simulation(rain, plant, temperature)

    samples = []
    result = cold_result
    for _ in range(repeats):
        gc.collect()
        elapsed, result = timed_simulation(rain, plant, temperature)
        samples.append(elapsed)

    middle = (days - 1) // 2
    print(
        f"Python {platform.python_implementation()} {platform.python_version()} "
        f"| days={days} | first_model_call_ms={cold_seconds * 1000:.3f} "
        f"| warmed_median_ms={statistics.median(samples) * 1000:.3f} "
        f"| repeats={repeats} | native_jit={native_jit_status}"
    )
    print(
        f"CHECK dry_days={result[1]} first={result[0][0]:.6f} "
        f"middle={result[0][middle]:.6f} last={result[0][-1]:.6f}"
    )


if __name__ == "__main__":
    main()

    using Printf

abstract type Plant end
struct Lettuce <: Plant end
struct Cactus <: Plant end

# Julia picks a threshold by dispatching on the plant's concrete type.
dry_limit(::Lettuce) = 5.0
dry_limit(::Cactus) = 2.0

function simulate(rain::Vector{Float64}, plant::T, temperature::Float64) where {T <: Plant}
    evaporation = 0.1temperature

    # Julia's dotted subtraction broadcasts across the array and creates a result vector.
    daily_change = rain .- evaporation

    soil = Vector{Float64}(undef, length(rain))
    soil[1] = clamp(10.0 + daily_change[1], 0.0, 20.0)
    for day in 2:length(rain)
        soil[day] = clamp(soil[day - 1] + daily_change[day], 0.0, 20.0)
    end

    limit = dry_limit(plant)
    dry_days = count(water -> water < limit, soil)
    return (soil=soil, dry_days=dry_days, within_limit=dry_days <= 2)
end

function timed_simulation(rain, plant, temperature)
    started = time_ns()
    result = simulate(rain, plant, temperature)
    elapsed = (time_ns() - started) / 1.0e9
    return elapsed, result
end

# Warm the timing wrapper and model before collecting steady-state samples.

for _ in 1:2
    GC.gc()
    timed_simulation(rain, plant, temperature)
end

samples = Vector{Float64}(undef, repeats)
for run in 1:repeats
    GC.gc()
    elapsed, _ = timed_simulation(rain, plant, temperature)
    samples[run] = elapsed
end
sort!(samples)
median_seconds = samples[(repeats + 1) ÷ 2]

middle = (days + 1) ÷ 2
@printf("Julia %s | days=%d | first_model_call_ms=%.3f | warmed_median_ms=%.3f | repeats=%d\n",
